cmake_minimum_required(VERSION 3.16)
project(SLeeLaImaging VERSION 2.9.0 LANGUAGES CXX)

add_executable(sleela-imaging native/imaging.cpp)
target_compile_features(sleela-imaging PRIVATE cxx_std_17)
set_target_properties(sleela-imaging PROPERTIES CXX_EXTENSIONS OFF)
if(MSVC)
  target_compile_options(sleela-imaging PRIVATE /W4 /WX /permissive-)
else()
  target_compile_options(sleela-imaging PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()

add_library(sleela_imaging_provider SHARED native/imaging_provider.cpp)

add_library(sleela_imaging_host SHARED native/imaging_host_adapter.cpp)
target_include_directories(sleela_imaging_host PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/native)
target_compile_features(sleela_imaging_host PRIVATE cxx_std_17)
target_compile_definitions(sleela_imaging_host PRIVATE SLI_HOST_ADAPTER_BUILD)
set_target_properties(sleela_imaging_host PROPERTIES CXX_EXTENSIONS OFF)
if(UNIX AND NOT APPLE)
  target_link_libraries(sleela_imaging_host PRIVATE dl)
endif()
if(MSVC)
  target_compile_options(sleela_imaging_host PRIVATE /W4 /WX /permissive-)
else()
  target_compile_options(sleela_imaging_host PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()

# SLeeLa's existing osRun/osSpawn VM built-ins can invoke this process adapter;
# the VM itself does not need a new opcode or a rebuilt runtime.
add_executable(sleela-imaging-bridge native/imaging_sleela_bridge.cpp)
target_include_directories(sleela-imaging-bridge PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/native)
target_link_libraries(sleela-imaging-bridge PRIVATE sleela_imaging_host)
target_compile_features(sleela-imaging-bridge PRIVATE cxx_std_17)
set_target_properties(sleela-imaging-bridge PROPERTIES CXX_EXTENSIONS OFF)
if(MSVC)
  target_compile_options(sleela-imaging-bridge PRIVATE /W4 /WX /permissive-)
else()
  target_compile_options(sleela-imaging-bridge PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()

target_include_directories(sleela_imaging_provider PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/native)
target_compile_features(sleela_imaging_provider PRIVATE cxx_std_17)
target_compile_definitions(sleela_imaging_provider PRIVATE SLI_PROVIDER_BUILD)
set_target_properties(sleela_imaging_provider PROPERTIES CXX_EXTENSIONS OFF)
if(MSVC)
  target_compile_options(sleela_imaging_provider PRIVATE /W4 /WX /permissive-)
else()
  target_compile_options(sleela_imaging_provider PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()

enable_testing()
find_package(Python3 REQUIRED COMPONENTS Interpreter)
add_test(NAME imaging-bridge-reviewed-plan COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/test_bridge_plan.py $<TARGET_FILE:sleela-imaging-bridge> $<TARGET_FILE:sleela_imaging_provider>)
add_test(NAME imaging-integration COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/test_imaging.py $<TARGET_FILE:sleela-imaging>)
add_executable(imaging-provider-tests tests/test_provider.cpp)
target_link_libraries(imaging-provider-tests PRIVATE sleela_imaging_provider)
target_compile_features(imaging-provider-tests PRIVATE cxx_std_17)
add_test(NAME imaging-provider-contract COMMAND imaging-provider-tests)
